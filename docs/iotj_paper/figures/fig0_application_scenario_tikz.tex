\begin{tikzpicture}[
    font=\footnotesize,
    >=Latex,
    node distance=0.38cm,

        align=center,

    },

\node[authbox, below=of capture] (auth) {OOB-guided RF-HSTU\\authentication};

\draw[arr] (capture) -- (auth);
\draw[arr] (auth) -- (accept);
